function doGet(e) {
  var sheet = getOrCreateSheet();
  var id = (e && e.parameter && e.parameter.id) || 'ihtifal_2026_utama';
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] == id) {
      return ContentService.createTextOutput(rows[i][1]).setMimeType(ContentService.MimeType.JSON);
    }
  }
  return ContentService.createTextOutput('{}').setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = getOrCreateSheet();
  var body = JSON.parse(e.postData.contents);
  var id = body.id || 'ihtifal_2026_utama';
  var jsonStr = JSON.stringify(body.data);
  var now = new Date().toLocaleString('ms-MY');
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0] == id) {
      sheet.getRange(i + 1, 2, 1, 2).setValues([[jsonStr, now]]);
      return ContentService.createTextOutput(JSON.stringify({status: 'ok'})).setMimeType(ContentService.MimeType.JSON);
    }
  }
  sheet.appendRow([id, jsonStr, now]);
  return ContentService.createTextOutput(JSON.stringify({status: 'ok'})).setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('DataPelan');
  if (!sheet) {
    sheet = ss.insertSheet('DataPelan');
    sheet.appendRow(['ID_Rekod', 'Data_JSON', 'Kemaskini_Terakhir']);
  }
  return sheet;
}
